const express = require('express');
const app = express();
const PORT = process.env.PORT || 3000;

// sample in-memory product list (replace with DB query)
const PRODUCTS = [
  { id:1, name:'Wireless Mouse', price:499, image:'https://picsum.photos/seed/mouse/200/200' },
  { id:2, name:'Mechanical Keyboard', price:2899, image:'https://picsum.photos/seed/keyboard/200/200' },
  { id:3, name:'Gaming Headset', price:1599, image:'https://picsum.photos/seed/headset/200/200' },
  { id:4, name:'USB-C Charger', price:799, image:'https://picsum.photos/seed/charger/200/200' },
  { id:5, name:'Laptop Stand', price:1199, image:'https://picsum.photos/seed/stand/200/200' }
];

app.get('/products', (req, res) => {
  const q = (req.query.q || '').toLowerCase().trim();
  if(!q) return res.json([]);

  // naive search: substring match in name
  const matches = PRODUCTS.filter(p => p.name.toLowerCase().includes(q));
  // optionally limit results
  res.json(matches.slice(0, 50));
});

app.listen(PORT, () => console.log('Server running on', PORT));
# Flask (Python)
from flask import Flask, request, jsonify
app = Flask(__name__)

PRODUCTS = [
  { 'id':1, 'name':'Wireless Mouse', 'price':499, 'image':'https://picsum.photos/seed/mouse/200/200' },
  { 'id':2, 'name':'Mechanical Keyboard', 'price':2899, 'image':'https://picsum.photos/seed/keyboard/200/200' },
  { 'id':3, 'name':'Gaming Headset', 'price':1599, 'image':'https://picsum.photos/seed/headset/200/200' }
]

@app.route('/products')
def products():
    q = (request.args.get('q') or '').lower().strip()
    if not q:
        return jsonify([])
    matches = [p for p in PRODUCTS if q in p['name'].lower()]
    return jsonify(matches[:50])

if __name__ == '__main__':
    app.run(port=3000, debug=True)
